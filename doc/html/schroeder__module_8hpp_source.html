69template<class R>
70SchroederVector<R> SchroederModule<R>::product(const SchroederVector<R>& ul, const SchroederVector<R>& ur, PType ptype) {
71 SchroederVector<R> res;
72 for (auto itl = ul.begin(); itl != ul.end(); ++ itl) {
73 for (auto itr = ur.begin(); itr != ur.end(); ++ itr) {
74 R r = itl -> second * itr -> second;
75 res.add(product(itl -> first, itr -> first, ptype), r);
76 }
77 }
78 return res;
79}

172 SchroederTensor<R> res;

174 const SchroederTree& t = it -> first;
