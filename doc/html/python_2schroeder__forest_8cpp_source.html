51int SchroederForestInit(SchroederForestObject* self, PyObject* args, PyObject* Py_UNUSED(ignored)) {
52 PyObject* ob = NULL;
53 if (PyArg_UnpackTuple(args, "SchroederTreenInit", 0, 1, &ob)) {
54 if (ob == NULL) {
55 // Call empty constructor
56 return 0;
57 }
58 PyTypeObject* type = Py_TYPE(ob);
59 if(strcmp(type -> tp_name, SchroederTreeType.tp_name) == 0) {
60 SchroederTreeObject* tree = (SchroederTreeObject*) ob;
61 self -> data = SchroederForest(tree -> data);
62 return 0;
63 }
64 PyErr_SetString(PyExc_AttributeError, "Argument must be a SchroederTree.");
65 return -1;
66 }
67 PyErr_SetString(PyExc_AttributeError, "At most one argument is required.");
68 return -1;
69}

76PyObject* SchroederForest_layer(SchroederForestObject* self, PyObject* args) {
77 PyObject* ob = NULL;
78 if(PyArg_UnpackTuple(args, "get", 1, 1, &ob)) {
79 if(PyLong_CheckExact(ob)) {
80 long i = PyLong_AsLong(ob);
81 if (i < 0 or i >= self -> data.height()) {
82 PyErr_SetString(PyExc_AttributeError, "Index out of range.");
83 return NULL;
84 }
85 return PyLong_FromLong(self -> data.layer(i));
86 }
87 PyErr_SetString(PyExc_AttributeError, "Argument must be an integer.");
88 return NULL;
89 }
90 PyErr_SetString(PyExc_AttributeError, "Exactly one argument is required.");
91 return NULL;
